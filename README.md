# Step 1: Download file from GEO

mkdir X_inactivation
prefetch  SRR16922266

fastq-dump SRR16922266 --split-files --gzip \
This will give you 3 files: _1,_2 and _3 
# Use zcat to check 
Example: SRR16922266_1.fastq.gz | head -20  \
SRR16922266_1.fastq.gz (Length 8, Sample Index or i7 index) \
SRR16922266_2.fastq.gz (Length 26, the Cell Barcode and UMI (Unique Molecular Identifier).\
SRR16922266_3.fastq.gz ( Length 91, the biological read)

# Step 2: Creating the Bam file (Either STAR or Cellranger)
Get the mm10 Cell Ranger reference
wget https://cf.10xgenomics.com/supp/cell-exp/refdata-cellranger-GRCh38-3.0.0.tar.gz

tar -zxvf refdata-cellranger-GRCh38-3.0.0.tar.gz

cellranger count --id=GSM7148 --transcriptome=refdata-cellranger-GRCh38-3.0.0  --fastqs=GSM7148/ --sample=sample

# Step 3: Create Conda environment and install

conda create -n haplo python=3.10 -y 

conda activate haplo  

conda install -c conda-forge -c bioconda -y \
samtools \
bcftools \
htslib \
pysam \
numpy \
pandas \
matplotlib \
tqdm \
whatshap


# Step 4: Execute Automated Pipeline Script : run_xci_pipeline.sh 
Use the STAR or Cellranger generated BAM file as input

chmod +x chrX_Haplo_Pipeline.sh


./chrX_Haplo_Pipeline.sh sample.bam sample
# Example:

./chrX_Haplo_Pipeline.sh GSM7148/possorted_genome_bam.bam GSM7148

It will generate output
| File                           | Description                                                 |
| ------------------------------ | ----------------------------------------------------------- |
| `GSM7148.clean.bam`            | Filtered BAM containing mapped reads.                       |
| `GSM7148.clean.sorted.bam`     | Coordinate-sorted BAM used for variant calling.             |
| `GSM7148.clean.sorted.bam.bai` | BAM index.                                                  |
| `GSM7148.raw.vcf.gz`           | Raw SNP calls generated by `bcftools`.                      |
| `GSM7148.filtered.vcf.gz`      | High-confidence variants after quality filtering.           |
| `GSM7148.het.vcf.gz`           | Heterozygous variants only.                                 |
| `chrX.het.vcf.gz`              | Heterozygous variants on chromosome X.                      |
| `chrX.phased.vcf.gz`           | WhatsHap-phased chromosome X variants.                      |
| `GSM7148_Hap0_vs_Hap1.png`     | Bar plot showing phased haplotype 0 and haplotype 1 counts. |
